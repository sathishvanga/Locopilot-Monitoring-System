## Streaming-to-disk (single-request) — memory-efficient, supports 1GB
Use this when client can send the whole file in one request. It streams the 
UploadFile to disk in large chunks and validates size on the fly.

```python
# streaming_upload.py
import os
import logging
import aiofiles
from fastapi import FastAPI, File, UploadFile, HTTPException, Request
from fastapi.middleware.cors import CORSMiddleware
from starlette.types import ASGIApp

logger = logging.getLogger("uvicorn.error")

ONE_GB = 1 * 1024 * 1024 * 1024
CHUNK_SIZE = 8 * 1024 * 1024  # 8MB per read

class LargeFileMiddleware:
    def __init__(self, app: ASGIApp, max_size: int = ONE_GB):
        self.app = app
        self.max_size = max_size

    async def __call__(self, scope, receive, send):
        if scope.get("type") == "http":
            scope.setdefault("state", {})["max_body_size"] = self.max_size
        await self.app(scope, receive, send)

app = FastAPI()
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
# Add middleware - adjust placement if your project already modifies 
middleware stack
app.add_middleware(LargeFileMiddleware, max_size=ONE_GB)

UPLOAD_DIR = "uploads"
os.makedirs(UPLOAD_DIR, exist_ok=True)

@app.post("/upload-video")
async def upload_video(file: UploadFile = File(...), request: Request = 
None, max_size: int = ONE_GB):
    filename = file.filename
    safe_filename = os.path.basename(filename)
    file_path = os.path.join(UPLOAD_DIR, safe_filename)
    total = 0

    # Optional: quick pre-check if Content-Length provided
    content_length = None
    if request:
        content_length = request.headers.get("content-length")
        if content_length:
            try:
                if int(content_length) > max_size:
                    raise HTTPException(status_code=413, detail=f"File 
exceeds {max_size // (1024*1024)}MB limit.")
            except ValueError:
                pass

    try:
        async with aiofiles.open(file_path, "wb") as out_file:
            while True:
                chunk = await file.read(CHUNK_SIZE)
                if not chunk:
                    break
                total += len(chunk)
                if total > max_size:
                    # cleanup partial
                    await out_file.close()
                    if os.path.exists(file_path):
                        os.remove(file_path)
                    logger.warning("Upload exceeded max size; removed 
partial file %s", file_path)
                    raise HTTPException(status_code=413, detail="File 
exceeds maximum allowed size.")
                await out_file.write(chunk)

        logger.info("Upload finished: %s (%d bytes)", file_path, total)
        return {"success": True, "filename": safe_filename, "size": total, 
"path": file_path}
    except HTTPException:
        raise
    except Exception as exc:
        logger.exception("Upload failed for %s", safe_filename)
        if os.path.exists(file_path):
            os.remove(file_path)
        raise HTTPException(status_code=500, detail=f"Upload failed: 
{str(exc)}")
````

## Server-managed chunked/resumable uploads (recommended for unreliable 
networks and very large files)

This gives clients a way to upload in parts and resume if interrupted. The 
server stores parts in a temporary folder, validates each part, and finally 
assembles them.

```python
# chunked_upload.py
import os
import uuid
import shutil
import math
import aiofiles
from typing import List
from fastapi import FastAPI, HTTPException, UploadFile, File, Form
from fastapi.responses import JSONResponse
from starlette.status import HTTP_200_OK

app = FastAPI()

TEMP_DIR = "uploads_tmp"         # holds chunk files per upload_id
FINAL_DIR = "uploads"            # assembled final files
os.makedirs(TEMP_DIR, exist_ok=True)
os.makedirs(FINAL_DIR, exist_ok=True)

MAX_SIZE = 1 * 1024 * 1024 * 1024  # 1GB

@app.post("/upload/initiate")
def initiate_upload(filename: str = Form(...), total_size: int = Form(...)):
    if total_size > MAX_SIZE:
        raise HTTPException(status_code=413, detail="Total size exceeds 
server limit.")
    upload_id = str(uuid.uuid4())
    upload_meta_dir = os.path.join(TEMP_DIR, upload_id)
    os.makedirs(upload_meta_dir, exist_ok=True)
    # Save metadata
    with open(os.path.join(upload_meta_dir, "meta.txt"), "w") as f:
        f.write(f"{filename}\n{total_size}\n")
    return {"upload_id": upload_id, "chunk_size_recommendation": 8 * 1024 * 
1024}

@app.post("/upload/chunk")
async def upload_chunk(
    upload_id: str = Form(...),
    part_number: int = Form(...),  # 1-based
    chunk: UploadFile = File(...)
):
    upload_path = os.path.join(TEMP_DIR, upload_id)
    if not os.path.exists(upload_path):
        raise HTTPException(status_code=404, detail="Upload ID not found.")
    chunk_filename = os.path.join(upload_path, 
f"part_{part_number:06d}.chunk")
    try:
        async with aiofiles.open(chunk_filename, "wb") as f:
            while True:
                data = await chunk.read(8 * 1024 * 1024)
                if not data:
                    break
                await f.write(data)
        return {"status": "ok", "part": part_number}
    except Exception as e:
        # cleanup partial chunk if exists
        if os.path.exists(chunk_filename):
            os.remove(chunk_filename)
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/upload/complete")
def complete_upload(upload_id: str = Form(...)):
    upload_path = os.path.join(TEMP_DIR, upload_id)
    meta_file = os.path.join(upload_path, "meta.txt")
    if not os.path.exists(meta_file):
        raise HTTPException(status_code=404, detail="Upload ID not found.")
    with open(meta_file, "r") as f:
        filename = f.readline().strip()
        total_size_expected = int(f.readline().strip())

    # list parts in order
    parts = sorted([p for p in os.listdir(upload_path) if 
p.startswith("part_")])
    if not parts:
        raise HTTPException(status_code=400, detail="No parts uploaded.")
    final_path = os.path.join(FINAL_DIR, os.path.basename(filename))
    bytes_written = 0
    try:
        with open(final_path, "wb") as dest:
            for part in parts:
                part_path = os.path.join(upload_path, part)
                with open(part_path, "rb") as pf:
                    while True:
                        chunk = pf.read(8 * 1024 * 1024)
                        if not chunk:
                            break
                        dest.write(chunk)
                        bytes_written += len(chunk)
        # verify assembled size
        if bytes_written != total_size_expected:
            # cleanup
            os.remove(final_path)
            raise HTTPException(status_code=400, detail=f"Assembled size 
mismatch: {bytes_written} != {total_size_expected}")
        # cleanup temp
        shutil.rmtree(upload_path)
        return {"success": True, "filename": filename, "size": 
bytes_written, "path": final_path}
    except Exception as e:
        if os.path.exists(final_path):
            os.remove(final_path)
        raise HTTPException(status_code=500, detail=str(e))
```

### Client-side chunk upload flow (high-level)

1. Call `/upload/initiate` with filename and total_size → receive 
`upload_id`.
2. Split file into N parts (e.g., 8MB each). Send each part as 
`multipart/form-data` to `/upload/chunk` with `upload_id` and `part_number`.
3. After all parts uploaded, call `/upload/complete` to assemble the file on
 the server.
4. Implement retry logic for failed part uploads and a status check if 
needed.

## Uvicorn/Gunicorn recommendations

* Uvicorn: `uvicorn main:app --host 0.0.0.0 --port 8000 --workers 1 
--timeout-keep-alive 120`

  * Use fewer workers for heavy disk I/O; consider running multiple 
instances behind a load balancer.
* Gunicorn + Uvicorn workers: `gunicorn -k uvicorn.workers.UvicornWorker 
--workers 2 --timeout 600 --graceful-timeout 120 main:app`
* Monitor worker memory and disk I/O; spawning many workers with heavy 
uploads will exhaust I/O.

## Logging, errors, and monitoring

* Return 413 for size checks, 404/400 for invalid upload_id or part missing,
 500 for server errors.
* Log start time, end time, bytes transferred, client IP, upload_id, and any
 exceptions.
* Track partial uploads for cleanup (e.g., remove temp folders older than 24
 hours via cron or background job).

## Housekeeping & operational notes

* Disk space: ensure disk has comfortable headroom — e.g., if N concurrent 
uploads possible, disk >= N * 1GB + spare.
* Throttling: limit concurrent full-backend uploads. Implement rate limits 
per client or concurrent-upload limits.
* Cleanup job: run a periodic task to remove temp dirs older than X hours 
(e.g., 24 hours).
* Integrity: optionally compute checksum (MD5/SHA256) on the client and 
verify after assemble.
* Security: validate filenames, sanitize input, and protect endpoints (auth 
/ signed tokens) if necessary.
* Resumable options: you can also adopt the TUS protocol via a TUS server if
 you want a standardized resumable flow.

     @router.post("/v2/upload/initiate")
     async def initiate_chunked_upload(
         filename: str = Form(...),
         total_size: int = Form(...),
         tripId: str = Form(...)
     ):
         # Validate size
         if total_size > settings.max_upload_size:
             raise HTTPException(status_code=413, detail="File exceeds 1GB 
     limit")

         # Check disk space
         has_space, err_msg = check_disk_space_available(
             settings.chunk_upload_dir, total_size, settings.disk_reserve_gb
         )
         if not has_space:
             raise HTTPException(status_code=507, detail=err_msg)

         # Create session
         upload_id, chunk_size = await 
     chunked_upload_service.initiate_upload(
             filename, total_size, tripId
         )

         return {
             "upload_id": upload_id,
             "chunk_size_recommendation": chunk_size,
             "total_chunks": math.ceil(total_size / chunk_size)
         }

     @router.post("/v2/upload/chunk")
     async def upload_chunk(
         upload_id: str = Form(...),
         part_number: int = Form(...),
         chunk: UploadFile = File(...)
     ):
         success = await chunked_upload_service.save_chunk(
             upload_id, part_number, chunk
         )
         if not success:
             raise HTTPException(status_code=404, detail="Upload session not
      found")

         return {"status": "ok", "part": part_number}

     @router.post("/v2/upload/complete")
     async def complete_chunked_upload(
         background_tasks: BackgroundTasks,
         upload_id: str = Form(...)
     ):
         try:
             # Assemble file
             file_path, total_size = await 
     chunked_upload_service.complete_upload(upload_id)

             # Extract metadata
             meta = chunked_upload_service.get_metadata(upload_id)
             trip_id = meta["trip_id"]

             # Process video (existing logic)
             result = video_processing_service.process_video(...)

             # Cleanup
             background_tasks.add_task(
                 video_processing_service.cleanup_uploaded_video,
                 file_path
             )
             background_tasks.add_task(
                 chunked_upload_service.cleanup_upload_session,
                 upload_id
             )

             return result

         except FileNotFoundError:
             raise HTTPException(status_code=404, detail="Upload session not
      found")
         except ValueError as e:
             raise HTTPException(status_code=400, detail=str(e))
         except Exception as e:
             logger.error(f"❌ Complete upload failed: {e}", exc_info=True)
             raise HTTPException(status_code=500, detail=str(e))
