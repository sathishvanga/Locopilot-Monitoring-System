  import os
  import cv2
  import json
  import torch
  import numpy as np
  from pathlib import Path
  from groundingdino.util.inference import load_model, predict
  from sam2.build_sam import build_sam2
  from sam2.sam2_image_predictor import SAM2ImagePredictor

  # --- Config ---
  FRAMES_DIR = "/Users/satishvanga/Documents/frames"
  OUTPUT_DIR = "/Users/satishvanga/Documents/labels"
  os.makedirs(OUTPUT_DIR, exist_ok=True)

  CLASSES = {
      "person sitting in train locomotive cabin": "person",
      "small rectangular mobile phone held in hand": "cell_phone",
      "open log book or register on desk surface": "book",
      "drinking cup or mug": "cup",
      "water bottle or drinking bottle": "bottle",
      "backpack or rucksack bag": "backpack",
      "small handbag or carry bag": "handbag",
      "suitcase or luggage bag": "suitcase",
      "handheld radio transceiver walkie talkie": "radio_handset",
  }

  CLASS_TO_ID = {v: i for i, v in enumerate(dict.fromkeys(CLASSES.values()))}
  TEXT_PROMPT = " . ".join(CLASSES.keys())

  # --- Load models ---
  grounding_model = load_model(
      "groundingdino/config/GroundingDINO_SwinT_OGC.py",
      "weights/groundingdino_swint_ogc.pth"
  )
  sam2 = build_sam2("sam2_hiera_l.yaml", "weights/sam2_hiera_large.pt")
  predictor = SAM2ImagePredictor(sam2)

  # --- Process frames ---
  for img_file in sorted(Path(FRAMES_DIR).glob("*.jpg")):
      image = cv2.imread(str(img_file))
      h, w = image.shape[:2]

      # Grounding DINO detection
      boxes, logits, phrases = predict(
          model=grounding_model,
          image=img_file,
          caption=TEXT_PROMPT,
          box_threshold=0.25,
          text_threshold=0.2,
      )

      # Convert to YOLO format and save
      label_file = Path(OUTPUT_DIR) / f"{img_file.stem}.txt"
      lines = []
      for box, phrase in zip(boxes, phrases):
          # Match phrase to class
          matched_class = None
          for prompt, cls_name in CLASSES.items():
              if phrase.lower() in prompt.lower() or
  prompt.lower().startswith(phrase.lower()):
                  matched_class = cls_name
                  break
          if matched_class is None:
              continue

          cx, cy, bw, bh = box.tolist()  # already normalized
          class_id = CLASS_TO_ID[matched_class]
          lines.append(f"{class_id} {cx:.6f} {cy:.6f} {bw:.6f} {bh:.6f}")

      label_file.write_text("\n".join(lines))
      print(f"{img_file.name}: {len(lines)} objects")

❯ instead of training can we shift to sam3 if performance better than yolo      

⏺ No — SAM 3 is not a replacement for YOLO in your pipeline. They solve
  different problems.

  Why SAM won't work here

  ┌────────────────┬────────────────────────┬───────────────────────────────┐
  │                │          YOLO          │             SAM 3             │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ Purpose        │ Object detection (bbox │ Segmentation (masks)          │
  │                │  + class)              │                               │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ Speed          │ ~5-10ms per frame      │ ~500-2000ms per frame         │
  │                │ (nano)                 │                               │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ Classification │ Built-in (outputs      │ None — needs Grounding DINO   │
  │                │ class + confidence)    │ to know what it found         │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ Prompting      │ Fully automatic        │ Requires text/point/box       │
  │                │                        │ prompt every frame            │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ GPU memory     │ ~500MB (nano)          │ ~4-8GB                        │
  ├────────────────┼────────────────────────┼───────────────────────────────┤
  │ Video pipeline │ Designed for it        │ Not designed for it           │
  └────────────────┴────────────────────────┴───────────────────────────────┘

  The real problems

  1. Your entire pipeline expects YOLO output — bounding boxes with class IDs
  and confidence scores flow into detect_objects(), ROI cropping, person
  proximity checks, temporal filtering, voting verification. SAM produces masks,
   not classified bboxes.
  2. SAM has no classifier — even with Grounding DINO, text-prompted detection
  is inconsistent frame-to-frame. You'd get flickering detections that break
  your temporal filtering logic.
  3. Speed at 0.5 FPS — you process 1 frame every 2 seconds. YOLO handles that
  in milliseconds. SAM + Grounding DINO would take 1-3 seconds per frame,
  leaving zero headroom for pose estimation, sleep detection, and all other
  processing.

  What you should actually do

  Train a custom YOLO11n on your locomotive cabin data. This gives you:
  - Same fast inference you have now
  - Much better accuracy on your specific camera angles and objects
  - Reliable cell_phone vs radio_handset separation (your biggest FP source)
  - Better small object detection for book and cup at CCTV resolution

  SAM is excellent for annotation (Option 1 from my previous answer), but YOLO
  is the right model for inference in a real-time video monitoring pipeline like
   yours.
